Vdc   = 600;          % [V] DC line voltage

Re    = 12;           % [ohm] excitation resistance
tau_e = 0.1;          % [s] excitation time constant
Le    = tau_e * Re;   % [H]

Ra    = 0.39;         % [ohm] armature resistance
tau_a = 10e-3;        % [s] armature time constant
La    = tau_a * Ra;   % [H]

K     = 1.06;         % [Nm/A^2] or [Vs/(A rad)] in this model
Ia_n  = 156;          % [A] overall rated armature current

% Mechanical data
mT = 15e3;            % [kg] empty tram mass
mp = 130 * 80;        % [kg] passengers
M  = mT + mp;         % [kg] total mass

d     = 0.680;        % [m] wheel diameter
r     = d/2;          % [m] wheel radius
rho   = 13/74;        % [-] gearbox ratio motor-to-wheel
beta  = 0.81;         % [Nms] friction coefficient

% Speeds
Omega_b = 970 * 2*pi/60;       % [rad/s] base/rated motor speed
v_b     = Omega_b * rho * r * 3.6;  % [km/h] base vehicle speed
vmax    = 42;                  % [km/h]
Omega_max = vmax/(3.6*r*rho);  % [rad/s]

% Equivalent inertia referred to motor shaft
J = M * rho^2 * d^2 / 4;

% Nominal Ratings and Limits
Pn    = 21e3;            % [W] rated power of a single motor
Ptot  = 4 * Pn;          % [W] total rated power (4 motors)
T_nom = Ptot / Omega_b;  % [Nm] nominal torque of the equivalent motor

% Rated back-emf (Updated to KVL method for anti-windup consistency)
En = Vdc - (Ra * Ia_n);  % [V]

% Conversion gains
kmh_to_rads = 1/(3.6*r*rho);
rads_to_kmh = 3.6*r*rho;
rads_to_ms  = r*rho;

slope_table = [0 0 5 0 0 -5 0 0];                    % [%]

%% PI Controller Tuning (Pole-Zero Cancellation)

% 1. Armature Current Controller
% Target bandwidth: 500 rad/s (approx 10 ms settling time)
wc_a = 900; 
Kp_a = La * wc_a;
Ki_a = Ra * wc_a;

% 2. Excitation Current Controller
% Target bandwidth: 100 rad/s (slightly slower than armature, sufficient for field weakening)
wc_e = 100;

Ki_e = Re * wc_e;

% 3. Speed Controller
% Target bandwidth: 10 rad/s (must be at least 10x slower than the inner loops for decoupling)
wc_w = 50;
Kp_w = J * wc_w;

%% 3. Overshoot Calculation for Speed Phases
disp('--- Transient Overshoot Analysis ---');

% Define the approximate start times of the step changes (from track profile)
% and the expected steady-state target velocities
step_times = [330, 825, 1000, 1490, 1815]; 
window_duration = 30; % seconds to look ahead for the peak

for i = 1:length(step_times)
    t_start = step_times(i);
    t_end = t_start + window_duration;
    
    % Extract data for this specific time window
    idx = find(t >= t_start & t <= t_end);
    t_window = t(idx);
    v_ref_window = v_ref(idx);
    v_meas_window = v_meas(idx);
    
    % Determine the step amplitude and target
    v_target = v_ref_window(end);
    v_initial = v_ref_window(1);
    step_amplitude = abs(v_target - v_initial);
    
    % Skip calculation if there is no actual step (e.g., flat track sections)
    if step_amplitude < 1
        continue; 
    end
    
    % Calculate Overshoot
    if v_target > v_initial % Accelerating
        v_peak = max(v_meas_window);
        overshoot_val = v_peak - v_target;
    else % Braking
        v_peak = min(v_meas_window);
        overshoot_val = v_target - v_peak; 
    end
    
    % Ensure overshoot is not negative (no overshoot case)
    overshoot_val = max(0, overshoot_val); 
    overshoot_pct = (overshoot_val / step_amplitude) * 100;
    
    fprintf('Step at t ≈ %d s | Amplitude: %5.2f km/h | Target: %5.2f km/h | Overshoot: %5.2f%%\n', ...
        round(t_start), step_amplitude, v_target, overshoot_pct);
end

%% 4. Disturbance Rejection at Slope Transitions
disp('--- Disturbance Rejection Analysis (Slope Transitions) ---');

% Reconstruct travelled position [km] by integrating measured speed
% (v_meas is in km/h, t is in seconds) -- skip this and use out.pos
% directly instead if your model already logs position.
pos_km = out.pos;

slope_change_dist = [3, 8];  % km: +5% uphill starts, -5% downhill starts
slope_labels = {'Uphill entry (+5%, km 3)', 'Downhill entry (-5%, km 8)'};

window_duration = 30; % seconds to look ahead for the peak deviation

for i = 1:length(slope_change_dist)
    t_event = interp1(pos_km, t, slope_change_dist(i), 'linear');
    
    if isnan(t_event)
        fprintf('%s: distance not reached in simulation.\n', slope_labels{i});
        continue;
    end
    
    idx = find(t >= t_event & t <= t_event + window_duration);
    v_ref_w  = v_ref(idx);
    v_meas_w = v_meas(idx);
    
    % v_ref is flat here, so any v_ref - v_meas gap is purely the
    % load-disturbance response, not a setpoint-tracking error.
    error_w = v_ref_w - v_meas_w;
    [peak_dev, ~] = max(abs(error_w));
    peak_pct = peak_dev / v_ref_w(1) * 100;
    
    fprintf('%s | t ~ %.1f s | peak deviation: %.3f km/h (%.2f%% of %.2f km/h)\n', ...
        slope_labels{i}, t_event, peak_dev, peak_pct, v_ref_w(1));
end
